import asyncio


def show_tasks():
    for i, task in enumerate(asyncio.all_tasks()):
        print(f'Task {i+1}: {task.get_name()}')
        print(f'  - Status:   {task._state}')
        print(f'  - Coroutine: {task.get_coro()}')
        print('-' * 40)


def cancel_task(task_name):
    for task in asyncio.all_tasks():
        if task.get_name() == task_name:
            print(f'trying to cancel task {task_name}')
            return task.cancel()

    def redraw(self):
        with hold_canvas(self.canvas):
            self.canvas.clear()
            s = f'{self.game.timer.displayed_time:06.2f}'
            self.canvas.fill_text(s, 30, 10)
            self.game.cannon.draw(self.canvas)
            for bullet in self.game.bullets:
                bullet.draw(self.canvas)

    def update(self, event, data):
        self.redraw()

def key_handler(self, key, state):
    timer = self.game.timer

        if timer.is_running:
            timer.is_running = False
        else:
