class Shape:

        self.width = width

    def move(self, dx, dy):
        x, y = self.anchor
        self.anchor = (x+dx, y+dy)

    def stroker(self, canvas, center, width):
        print(f'stroke {type(self).__name__} on canvas with center/corner {center} and widht/radius {width}')

    def filler(self, canvas, center, width):
        print(f'stroke {type(self).__name__} on canvas at center/corner {center} and widht/radius {width}')

    def draw(self, canvas, clear=True,  fill_style=None, stroke_style='black'):
        canvas.save()

        if clear:
            canvas.clear()
        if fill_style:
            canvas.fill_style = fill_style
            self.filler(canvas, self.anchor, self.width)
        if stroke_style:
            canvas.stroke_style = stroke_style
            self.stroker(canvas, self.anchor, self.width)

        canvas.restore()
