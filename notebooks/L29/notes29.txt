RANK_VALUE = ({r: int(r) for r in Deck.ranks[:8]} |
              dict(zip('TJQKA', (10, 10, 10, 10, 11)))
              )

RANK_VALUE

def handvalue(hand):
    value = sum(RANK_VALUE[card[0]] for card in hand)
    n_aces = sum(card[0] == 'A' for card in hand)

    while value > 21 and n_aces > 0:
        value -= 10
        n_aces -= 1
    return value

class BlackJack:
    def __init__(self):
        self.deck = Deck()
        self.deck.shuffle()
        self.player = []  # omit, show _repr__ problem

    def new_game(self):

            self.deck = Deck()
            self.deck.shuffle()

        self.player = self.deck.deal(2)
        self.dealer = self.deck.deal(2)

        print(self)
        if handvalue(self.player) == 21:
            print('player wins with BlackJack!')

    def hit(self):
        if self.player:
            self.player.append(self.deck.deal())
        if handvalue(self.player) >= 21:
            print(f'player loses: {self}')
        else:
            print(f'player: {self.player}')

    def stand(self):
        if self.dealer:
            while handvalue(self.dealer) < 17:
                self.dealer.append(self.deck.deal())

            val_p = handvalue(self.player)
            val_d = handvalue(self.dealer)

            if val_d >= 21 or val_p > val_d:
                print('player wins')
            elif val_p == val_d:
                print('draw')
            else:
                print('player loses')
            print(f'player: {val_p}, dealer: {val_d}, {self.dealer}')

    def __repr__(self):
