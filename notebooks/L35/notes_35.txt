class View:
    def __init__(self, game):
        self.game = game
        self.canvas = W.get_canvas(600, 180)

        game.register_callback(self.update)
        self.update(game.numbers)


    def update(self, numbers):
        with hold_canvas():
            self.canvas.clear()

            self.canvas.fill_style = 'white'
            self.canvas.fill_rect(0, 0, 600, 180)

            for i in range(len(numbers)):
                row = i // 10
                column = i % 10

                x = 20 + column * 55
                y = 40 + row * 60

                self.canvas.fill_style = 'blue'
                self.canvas.fill_rect(x, y, 50, 40)

                self.canvas.stroke_style = 'black'
                self.canvas.stroke_rect(x, y, 50, 40)

                self.canvas.fill_style = 'black'
                self.canvas.font = '16px Arial'
                self.canvas.text_align = 'center'
                self.canvas.text_baseline = 'middle'
                self.canvas.fill_text(str(numbers[i]), x + 25, y + 20)

            self.canvas.stroke_style = 'red'
            self.canvas.stroke_rect(17, 37, 221, 46)

    @notify
    def reveal_cell(self, row, col):
        '''
        Deckt ein Feld auf und behandelt Gewinn bzw. Game Over.
        '''
        grid = None
        flags_to_place = []
        
        if self.game_over:
            log('Ignoriert: Spiel ist bereits beendet.')
            return

        if self.grid.visible[row][col]:
            log(f'Ignoriert: Feld ({row}, {col}) ist bereits sichtbar.')
            return

        if self.grid.flags[row][col]:
            log(f'Ignoriert: Feld ({row}, {col}) ist markiert (Flagge).')
            return

        grid = None
        if not self.grid.initialized:
            self.grid.initialize(row, col)
            # self.view.draw_background(self.grid)  #
            grid = self.grid
            self.timer.start()
            self.flag_mode_button.disabled = False
            log(f'Spielfeld initialisiert. Erstes Feld: ({row}, {col})')

        revealed = self.grid.reveal(row, col)
        
        # self.view.reveal_cells(revealed)
        cells_to_reveal = revealed
        
        log(f'Feld aufgedeckt: ({row}, {col}), Mine: {self.grid.mines[row][col]}')

        # Verloren: Mine getroffen: Spiel beenden und alle Minen aufdecken.
        if self.grid.mines[row][col]:
            self.game_over = True
            self.timer.stop()
            revealed_mines = self.grid.reveal_all_mines()
            # self.view.reveal_cells(revealed_mines)
            cells_to_reveal += revealed_mines
            self.output_area.clear_output()
            self.show_game_over_message()
            # return

        # Gewonnen: Alle sicheren Felder aufgedeckt: Spiel beenden.
        elif self.grid.check_win():
            self.game_over = True
            self.timer.stop()
            self.update_highscore()
            flagged = self.grid.flag_all_mines()
            for flag_row, flag_col in flagged:
                # self.view.set_flag(flag_row, flag_col)  #
                flag = (flag_row, flag_col)
                flags_to_place.append(flag)
            self.output_area.clear_output()
            self.show_win_message()

        return cells_to_reveal, flags_to_place, grid
